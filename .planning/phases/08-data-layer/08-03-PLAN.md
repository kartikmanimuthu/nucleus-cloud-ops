    name: "physical-name",
    userPoolId: userPool.id,
    generateSecret: true,

    logoutUrls: ["http://localhost:3000", appUrl],
    preventUserExistenceErrors: "ENABLED",
    enableTokenRevocation: true,
    accessTokenValidity: 1,   // hours
    idTokenValidity: 1,       // hours
    refreshTokenValidity: 30, // days
    tokenValidityUnits: {
        accessToken: "hours",
        idToken: "hours",
        refreshToken: "days",
    },
    supportedIdentityProviders: ["COGNITO"],
});

const identityPool = new aws.cognito.IdentityPool("name", {
    identityPoolName: "physical-name",
    allowUnauthenticatedIdentities: false,

        providerName: userPool.endpoint,
    }],
});

const authenticatedRole = new aws.iam.Role("name", {
    name: "physical-name",
    assumeRolePolicy: identityPool.id.apply(poolId =>

            Version: "2012-10-17",
            Statement: [{
                Effect: "Allow",
                Principal: { Federated: "cognito-identity.amazonaws.com" },
                Action: "sts:AssumeRoleWithWebIdentity",
                Condition: {
                    StringEquals: { "cognito-identity.amazonaws.com:aud": poolId },
                    "ForAnyValue:StringLike": { "cognito-identity.amazonaws.com:amr": "authenticated" },
                },
            }],
        })
    ),
});

new aws.cognito.IdentityPoolRoleAttachment("name", {
    identityPoolId: identityPool.id,
    roles: { authenticated: authenticatedRole.arn },
