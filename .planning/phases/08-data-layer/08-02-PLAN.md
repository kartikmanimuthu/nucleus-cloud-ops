From @pulumi/aws S3 pattern:
```typescript
const bucket = new aws.s3.BucketV2("name", {
    bucket: "physical-name",
    forceDestroy: false,
}, { retainOnDelete: true });

new aws.s3.BucketLifecycleConfigurationV2("name-lifecycle", {
    bucket: bucket.id,
    rules: [{
        id: "rule-id",
        status: "Enabled",
        filter: { prefix: "raw/" },  // or {} for all objects
        expiration: { days: 365 },
    }],
});
```

From @pulumi/aws SQS pattern:
```typescript
const dlq = new aws.sqs.Queue("dlq-name", {
    name: "physical-dlq-name",
    messageRetentionSeconds: 1209600,
});

const queue = new aws.sqs.Queue("queue-name", {
    name: "physical-queue-name",
    visibilityTimeoutSeconds: 900,
    receiveWaitTimeSeconds: 20,
    redrivePolicy: JSON.stringify({
        deadLetterTargetArn: dlq.arn,
        maxReceiveCount: 3,
    }),
});

// Separate resource for queue policy (not inline)
new aws.sqs.QueuePolicy("policy-name", {
    queueUrl: queue.url,
    policy: pulumi.all([queue.arn, bucket.arn]).apply(([queueArn, bucketArn]) =>
        JSON.stringify({
            Version: "2012-10-17",
            Statement: [{
                Effect: "Allow",
                Principal: { Service: "s3.amazonaws.com" },
                Action: "sqs:SendMessage",
                Resource: queueArn,
                Condition: { ArnLike: { "aws:SourceArn": bucketArn } },
            }],
        })
    ),
});
```

From @pulumi/aws CloudWatch pattern:
```typescript
new aws.cloudwatch.MetricAlarm("alarm-name", {
    alarmName: "physical-alarm-name",
    alarmDescription: "description",
    namespace: "AWS/SQS",
    metricName: "ApproximateNumberOfMessagesVisible",
    dimensions: { QueueName: dlq.name },
    statistic: "Sum",
    period: 300,
    evaluationPeriods: 1,
    threshold: 1,
    comparisonOperator: "GreaterThanOrEqualToThreshold",
    treatMissingData: "notBreaching",
});
```
