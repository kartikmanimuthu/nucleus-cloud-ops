       ```typescript
       async function handleKbSync(jobData: unknown): Promise<void> {
         const data = jobData as KBSyncJob;
         const { kbId, dsId } = data;

         log.info('Processing sync', { type: data.type, kbId, dsId });

         const ds = await getDataSource(kbId, dsId);
         const oldVectorCount = (ds?.vectorCount as number) || 0;
         const oldVectorKeys: string[] = data.oldVectorKeys || (ds?.vectorKeys as string[]) || [];

         try {
           if (oldVectorKeys.length) {
             await deleteOldVectors(oldVectorKeys);
             await updateKBVectorCount(kbId, -oldVectorCount);
           }

           let vectorKeys: string[];
           switch (data.type) {
             case 'file-upload':     vectorKeys = await handleFileUpload(data); break;
             case 's3-sync':         vectorKeys = await handleS3Sync(data); break;
             case 'confluence-sync': vectorKeys = await handleConfluenceSync(data); break;
             case 'bitbucket-sync':  vectorKeys = await handleBitbucketSync(data); break;
             default: throw new Error(`Unknown job type: ${(data as KBSyncJob).type}`);
           }

           await updateDS(kbId, dsId, {
             status: 'synced',
             vectorCount: vectorKeys.length,
             vectorKeys,
             lastSyncAt: new Date().toISOString(),
             lastSyncError: null,
           });
           await updateKBVectorCount(kbId, vectorKeys.length);

           log.info('Sync complete', { type: data.type, kbId, dsId, vectorCount: vectorKeys.length });
         } catch (err) {
           const shortMessage = err instanceof Error ? err.message : 'Sync failed';
           const fullDetail = err instanceof Error
             ? `${err.message}\n${err.stack ?? ''}`
             : String(err);
           log.error('Sync failed', { type: data.type, kbId, dsId, error: shortMessage });
           try {
             await updateDS(kbId, dsId, {
               status: 'error',
               lastSyncError: shortMessage,
               lastErrorMessage: shortMessage,
               lastErrorDetail: fullDetail,
             });
           } catch (e) {
             log.error('Status update failed', { kbId, dsId, error: e instanceof Error ? e.message : String(e) });
           }
           throw err;
         }
       }
       ```
