    ```typescript
    import { Module, Action, PredefinedRole, RoleLevel, PermissionSet } from './types';

    export const ROLE_PERMISSIONS: Record<PredefinedRole, PermissionSet> = {
      Owner: {
        Accounts: ['create', 'read', 'update', 'delete'],
        Schedules: ['create', 'read', 'update', 'delete'],
        AIOps: ['create', 'read', 'update', 'delete'],
        Inventory: ['create', 'read', 'update', 'delete'],
        Settings: ['create', 'read', 'update', 'delete'],
      },
      Admin: {
        Accounts: ['create', 'read', 'update', 'delete'],
        Schedules: ['create', 'read', 'update', 'delete'],
        AIOps: ['create', 'read', 'update', 'delete'],
        Inventory: ['create', 'read', 'update', 'delete'],
        Settings: ['create', 'read', 'update'],  // No delete on Settings per D-04
      },
      Member: {
        Accounts: ['create', 'read', 'update'],
        Schedules: ['create', 'read', 'update'],
        AIOps: ['create', 'read', 'update'],
        Inventory: ['create', 'read', 'update'],
        Settings: ['read'],  // R only on Settings per D-04
      },
      Viewer: {
        Accounts: ['read'],
        Schedules: ['read'],
        AIOps: ['read'],
        Inventory: ['read'],
        Settings: ['read'],
      },
    };

    export const ROLE_LEVELS: Record<PredefinedRole, RoleLevel> = {
      Owner: 4,
      Admin: 3,
      Member: 2,
      Viewer: 1,
    };

    // Minimum level to assign roles (per D-09: Member and Viewer cannot assign)
    const MIN_ASSIGN_LEVEL: RoleLevel = 3; // Admin

    export function hasPermission(role: PredefinedRole, action: Action, module: Module): boolean {
      const perms = ROLE_PERMISSIONS[role];
      if (!perms) return false;
      return perms[module]?.includes(action) ?? false;
    }

    // Check custom permission set (for DB-stored custom roles)
    export function hasCustomPermission(permissions: PermissionSet, action: Action, module: Module): boolean {
      return permissions[module]?.includes(action) ?? false;
    }

    // Per D-09: can assignerRole assign targetRole?
    export function canAssignRole(assignerRole: PredefinedRole, targetRole: PredefinedRole): boolean {
      const assignerLevel = ROLE_LEVELS[assignerRole];
      const targetLevel = ROLE_LEVELS[targetRole];
      if (assignerLevel < MIN_ASSIGN_LEVEL) return false; // Member/Viewer can't assign
      return assignerLevel >= targetLevel;
    }

    // Per D-10: auto-level custom roles by permission count
    export function getAutoLevel(permissions: PermissionSet): RoleLevel {
      const totalActions = Object.values(permissions).flat().length;
      const maxPossible = 20; // 5 modules * 4 actions
      if (totalActions >= maxPossible) return 4; // Owner-level
      if (totalActions >= 15) return 3; // Admin-level
      if (totalActions >= 8) return 2; // Member-level
      return 1; // Viewer-level
    }
    ```

    The new authorize() function:
    1. Accepts the SAME signature: `authorize(action: Actions | Action, subjectType: Subjects | Module, subjectData?: Record<string, any>)`
    2. Checks `process.env.USE_NEW_RBAC` env var (single global flag for now — per-route flags are overkill since we'll migrate all routes in one plan)
    3. If `USE_NEW_RBAC !== 'true'`, delegate to old CASL path (import getServerAbility from `./server-ability`)
    4. If `USE_NEW_RBAC === 'true'`, use new path:
       a. Call `getAuthSession()` from `@/lib/auth-session`
       b. If no session or no user, return 401 NextResponse
       c. Extract `role` from `session.user.role`
       d. If role is a PredefinedRole ('Owner'|'Admin'|'Member'|'Viewer'), call `hasPermission(role, mappedAction, mappedModule)`
       e. If role is not predefined (custom role), look up custom role permissions from DB via a new `getCustomRolePermissions(roleName, tenantId)` function (stub it for now — Plan 03 implements the real DB lookup; for now return empty PermissionSet which means deny-all for unknown custom roles)
       f. Map old subject names to modules using SUBJECT_TO_MODULE from types.ts
       g. Map old action names using ACTION_MAP from types.ts
       h. If permission denied, return `NextResponse.json({ error: 'Forbidden', message: \`You do not have permission to ${action} ${subjectType}\` }, { status: 403 })`
       i. If permitted, return null

    Keep the existing `isAdmin()`, `can()`, `cannot()` exports but update them to also respect the feature flag.
