```typescript
import { describe, it, expect, vi, beforeEach } from 'vitest';
import { NextRequest } from 'next/server';

// Mock auth session — controls which tenant the test impersonates
vi.mock('@/lib/auth-session', () => ({
    getSessionTenantId: vi.fn(),
    getAuthSession: vi.fn(),
    getSessionUserId: vi.fn(),
}));

// Mock RBAC — always allow (we're testing data isolation, not permissions)
vi.mock('@/lib/rbac/authorize', () => ({
    authorize: vi.fn().mockResolvedValue(null),
}));

// Mock the repository factory — return a mock repo that tracks tenantId
vi.mock('@/lib/db/repository-factory', () => ({
    get{Module}Repository: vi.fn(),
}));

import { getSessionTenantId } from '@/lib/auth-session';
import { get{Module}Repository } from '@/lib/db/repository-factory';
import { GET } from '@/app/api/{module}/route';

describe('{Module} API — cross-tenant isolation', () => {
    let mockRepo: { getXxx: ReturnType<typeof vi.fn> };

    beforeEach(() => {
        vi.clearAllMocks();
        mockRepo = {
            getXxx: vi.fn().mockResolvedValue({ items: [], totalCount: 0 }),
        };
        vi.mocked(get{Module}Repository).mockReturnValue(mockRepo as any);
        vi.mocked(getSessionTenantId).mockResolvedValue('tenant-a');
    });

    it('GET list passes tenant-a to repository — tenant-b data never queried', async () => {
        const req = new NextRequest('http://localhost:3000/api/{module}');
        await GET(req);

        // The repository method must have been called with tenant-a
        expect(mockRepo.getXxx).toHaveBeenCalledWith(
            expect.objectContaining({ tenantId: 'tenant-a' })
        );
        // Verify tenant-b was never passed
        const calls = mockRepo.getXxx.mock.calls;
        for (const call of calls) {
            const arg = call[0];
            if (typeof arg === 'object' && arg.tenantId) {
                expect(arg.tenantId).not.toBe('tenant-b');
            }
        }
    });

    it('switching session to tenant-b queries tenant-b only', async () => {
        vi.mocked(getSessionTenantId).mockResolvedValue('tenant-b');
        const req = new NextRequest('http://localhost:3000/api/{module}');
        await GET(req);

        expect(mockRepo.getXxx).toHaveBeenCalledWith(
            expect.objectContaining({ tenantId: 'tenant-b' })
        );
    });
});
```
