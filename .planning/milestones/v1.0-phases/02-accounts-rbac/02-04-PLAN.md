**DynamoDB query pattern (GSI1 pagination):**
```typescript
const queryAccounts = async (): Promise<DynamoAccountItem[]> => {
  const items: DynamoAccountItem[] = [];
  let lastKey: Record<string, unknown> | undefined;
  do {
    const result = await dynamoClient.send(new QueryCommand({
      TableName: APP_TABLE_NAME,
      IndexName: 'GSI1',
      KeyConditionExpression: 'gsi1pk = :gsi1pk',
      ExpressionAttributeValues: { ':gsi1pk': 'TYPE#ACCOUNT' },
      ExclusiveStartKey: lastKey,
    }));
    items.push(...(result.Items as DynamoAccountItem[] ?? []));
    lastKey = result.LastEvaluatedKey as Record<string, unknown> | undefined;
  } while (lastKey);
  return items;
};
```

**Upsert pattern:**
```typescript
const migrateAccounts = async () => {
  const items = await queryAccounts();
  console.log(`Found ${items.length} account records in DynamoDB`);
  let migrated = 0;
  for (const item of items) {
    const tenantId = item.tenantId ?? item.pk?.replace('TENANT#', '') ?? 'org-default';
    const accountId = item.accountId ?? item.sk?.replace('ACCOUNT#', '');
    if (!accountId) { console.warn('Skipping item with no accountId:', item.pk, item.sk); continue; }

    await prisma.account.upsert({
      where: { tenantId_accountId: { tenantId, accountId } },
      update: {
        name: item.accountName ?? accountId,
        roleArn: item.roleArn ?? '',
        externalId: item.externalId ?? null,
        regions: item.regions ?? [],
        active: item.active ?? true,
        description: item.description ?? null,
        connectionStatus: item.connectionStatus ?? 'unknown',
        connectionError: item.connectionError ?? null,
        updatedAt: item.updatedAt ? new Date(item.updatedAt) : new Date(),
        updatedBy: item.updatedBy ?? 'migration',
      },
      create: {
        tenantId,
        accountId,
        name: item.accountName ?? accountId,
        roleArn: item.roleArn ?? '',
        externalId: item.externalId ?? null,
        regions: item.regions ?? [],
        active: item.active ?? true,
        description: item.description ?? null,
        connectionStatus: item.connectionStatus ?? 'unknown',
        connectionError: item.connectionError ?? null,
        createdAt: item.createdAt ? new Date(item.createdAt) : new Date(),
        updatedAt: item.updatedAt ? new Date(item.updatedAt) : new Date(),
        createdBy: item.createdBy ?? 'migration',
        updatedBy: item.updatedBy ?? 'migration',
      },
    });
    migrated++;
    console.log(`Migrated ${migrated}/${items.length} records... (accountId: ${accountId})`);
  }
  console.log(`\nMigration complete. Migrated ${migrated}/${items.length} account records.`);
};
```
