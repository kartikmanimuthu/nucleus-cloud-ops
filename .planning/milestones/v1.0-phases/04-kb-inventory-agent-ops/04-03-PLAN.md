```python
"""
PostgreSQL writer for discovery Lambda.
Writes inventory resources to the inventory_resources table using psycopg2.
Controlled by USE_PG_INVENTORY env var (dual-write alongside DynamoDB).
"""
import os
import json
import logging
from datetime import datetime, timezone
from typing import List, Dict, Any, Optional

logger = logging.getLogger(__name__)

def is_pg_enabled() -> bool:
    return os.environ.get('USE_PG_INVENTORY', 'false').lower() == 'true'

def get_connection():
    """Create a psycopg2 connection from DATABASE_URL."""
    import psycopg2
    database_url = os.environ.get('DATABASE_URL')
    if not database_url:
        raise ValueError("DATABASE_URL environment variable is required for PostgreSQL writes")
    return psycopg2.connect(database_url)

def write_resources_to_pg(
    resources: List[Dict[str, Any]],
    tenant_id: str,
    account_id: str,
    batch_size: int = 500
) -> int:
    """
    Upsert discovered resources to PostgreSQL inventory_resources table.
    Uses ON CONFLICT (tenant_id, account_id, resource_type, resource_id) DO UPDATE.
    Batched in chunks of batch_size for memory efficiency.
    Returns count of upserted records.
    """
    if not resources:
        return 0

    conn = get_connection()
    total = 0
    try:
        with conn.cursor() as cur:
            for i in range(0, len(resources), batch_size):
                batch = resources[i:i + batch_size]
                values = []
                params = []
                for r in batch:
                    # Generate a cuid-like id (use resource composite key hash)
                    resource_type = r.get('resourceType', r.get('resource_type', ''))
                    resource_id = r.get('resourceId', r.get('resource_id', ''))
                    region = r.get('region', '')
                    name = r.get('name', r.get('resourceName', None))
                    status = r.get('status', r.get('state', None))
                    tags = json.dumps(r.get('tags', {}))
                    # Everything not in typed columns goes to metadata JSONB (per D-03)
                    metadata = json.dumps(r.get('metadata', r.get('details', {})))
                    now = datetime.now(timezone.utc).isoformat()

                    values.append("(%s, %s, %s, %s, %s, %s, %s, %s, %s, %s, NOW(), NOW())")
                    params.extend([
                        tenant_id, account_id, region, resource_type,
                        resource_id, name, status, tags, metadata, now
                    ])

                if not values:
                    continue

                sql = f"""
                    INSERT INTO inventory_resources
                        (tenant_id, account_id, region, resource_type, resource_id,
                         name, status, tags, metadata, discovered_at, updated_at)
                    VALUES {', '.join(values)}
                    ON CONFLICT (tenant_id, account_id, resource_type, resource_id)
                    DO UPDATE SET
                        name = EXCLUDED.name,
                        status = EXCLUDED.status,
                        tags = EXCLUDED.tags,
                        metadata = EXCLUDED.metadata,
                        updated_at = NOW()
                """
                # Note: Prisma uses camelCase column names by default, but @@map
                # maps the model to snake_case table. Check actual column names
                # from the migration SQL. Prisma schema field names ARE the column
                # names unless @map is used. Since we used camelCase fields
                # (tenantId, accountId, etc.) without @map, the actual PostgreSQL
                # columns will be "tenantId", "accountId" etc. (quoted identifiers).
                # IMPORTANT: Read the actual migration SQL to confirm column names
                # before finalizing this query. Adjust column names to match.

                cur.execute(sql, params)
                total += len(batch)
                logger.info(f"[pg_writer] Upserted batch {i//batch_size + 1}: {len(batch)} resources")

            conn.commit()
    except Exception as e:
        conn.rollback()
        logger.error(f"[pg_writer] Failed to write resources to PostgreSQL: {e}")
        raise
    finally:
        conn.close()

    logger.info(f"[pg_writer] Total upserted: {total} resources for account {account_id}")
    return total
```
