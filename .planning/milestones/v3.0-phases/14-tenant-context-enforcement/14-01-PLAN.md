**1. Add `getTenantClient(tenantId)` to `web-ui/lib/db/pg-config.ts` (per D-01, D-02, D-03):**

Add this function alongside the existing `getPrismaClient()`:

```typescript
import { Prisma } from '@prisma/client';

const TENANT_SCOPED_MODELS = new Set([
    'Account', 'Schedule', 'ScheduleExecution', 'TargetedResource',
    'AuditLog', 'KnowledgeBase', 'DataSource', 'InventoryResource',
    'AgentOpsRun', 'AgentOpsEvent', 'ScheduledTask', 'AgentMemory',
    'ChatMessage', 'CustomRole', 'UserTenantRole', 'TenantConfig',
]);

export function getTenantClient(tenantId: string) {
    if (!tenantId) throw new Error('getTenantClient: tenantId is required');
    const base = getPrismaClient();
    return base.$extends({
        query: {
            $allModels: {
                async $allOperations({ model, operation, args, query }) {
                    if (!TENANT_SCOPED_MODELS.has(model ?? '')) {
                        return query(args);
                    }
                    // Reads: inject tenantId into WHERE
                    if (['findMany', 'findFirst', 'findUnique', 'findUniqueOrThrow', 'count', 'aggregate', 'groupBy'].includes(operation)) {
                        args.where = { ...args.where, tenantId };
                    }
                    // Creates: inject tenantId into data
                    if (['create'].includes(operation)) {
                        args.data = { ...args.data, tenantId };
                    }
                    if (['createMany'].includes(operation)) {
                        if (Array.isArray(args.data)) {
                            args.data = args.data.map((d: any) => ({ ...d, tenantId }));
                        } else {
                            args.data = { ...args.data, tenantId };
                        }
                    }
                    if (['upsert'].includes(operation)) {
                        args.where = { ...args.where, tenantId };
                        args.create = { ...args.create, tenantId };
                        args.update = { ...args.update };
                    }
                    // Updates: inject tenantId into WHERE
                    if (['update', 'updateMany'].includes(operation)) {
                        args.where = { ...args.where, tenantId };
                    }
                    // Deletes: inject tenantId into WHERE
                    if (['delete', 'deleteMany'].includes(operation)) {
                        args.where = { ...args.where, tenantId };
                    }
                    return query(args);
                },
            },
        },
    });
}
```

Key points per D-01: Uses `$extends` wrapping the singleton from `getPrismaClient()`. Per D-02: Raw SQL (`$executeRaw`, `$queryRawUnsafe`) is NOT intercepted — callers must manually scope. Per D-03: Created per-request, no caching.

Export `TENANT_SCOPED_MODELS` as well (the test plan needs it).

model Tenant {
