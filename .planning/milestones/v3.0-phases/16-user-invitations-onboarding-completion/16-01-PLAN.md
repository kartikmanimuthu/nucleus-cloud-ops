model Invitation {
  id         String   @id @default(cuid())
  tenantId   String
  email      String
  role       String
  invitedBy  String
  status     String   @default("pending") // pending|accepted|revoked|expired
  createdAt  DateTime @default(now())
  expiresAt  DateTime // 7 days from creation (Cognito TemporaryPasswordValidityDays)

  @@unique([tenantId, email, status]) // prevent duplicate pending invites
  @@index([tenantId])
  @@index([tenantId, status])
  @@index([email])
  @@map("invitations")
}

Implementation details for each method:

**createInvitation:**
1. Check if email already has a pending invitation for this tenant: `prisma.invitation.findFirst({ where: { tenantId, email, status: 'pending' } })`. If exists, throw Error("An invitation is already pending for this email").
2. Check if user already a member: `prisma.userTenantRole.findFirst({ where: { tenantId, email } })`. If exists, throw Error("This user is already a member of your organization").
3. Check if user exists in AuthUser table: `getPrismaClient().authUser.findUnique({ where: { email } })`.
4. If user exists (D-08 auto-join path):
   - Create UserTenantRole immediately: `getPrismaClient().userTenantRole.create({ data: { userId: existingUser.id, tenantId, email, role, assignedAt: new Date(), assignedBy: invitedBy } })`
   - Create invitation record with status='accepted': `prisma.invitation.create({ data: { tenantId, email, role, invitedBy, status: 'accepted', expiresAt: new Date(Date.now() + 7*24*60*60*1000) } })`
   - Return `{ invitation, autoJoined: true }`
5. If user does NOT exist (D-04 new user path):
   - Call Cognito AdminCreateUser: `getCognitoClient().send(new AdminCreateUserCommand({ UserPoolId: COGNITO_USER_POOL_ID, Username: email, UserAttributes: [{ Name: 'email', Value: email }, { Name: 'email_verified', Value: 'true' }], DesiredDeliveryMediums: ['EMAIL'] }))`
   - Create invitation record with status='pending', expiresAt = 7 days from now
   - Return `{ invitation, autoJoined: false }`

Use `getTenantClient(tenantId)` for invitation queries (tenant-scoped). Use `getPrismaClient()` for AuthUser and UserTenantRole queries (platform-level, since UserTenantRole needs cross-tenant check for existing membership and AuthUser is not tenant-scoped).

**listInvitations:**
1. Fetch all invitations: `prisma.invitation.findMany({ where: { tenantId }, orderBy: { createdAt: 'desc' } })`
2. For each invitation where status='pending' and expiresAt < now, update status to 'expired' in-place
3. Return the list

**resendInvitation:**
1. Find invitation by id (tenant-scoped): must be status='pending'
2. Call AdminCreateUser with `MessageAction: 'RESEND'`
3. Update expiresAt to 7 days from now

**revokeInvitation:**
1. Find invitation by id (tenant-scoped): must be status='pending'
2. Try AdminDisableUser (catch if user doesn't exist — may have been cleaned up)
3. Update status to 'revoked'

**acceptPendingInvitation:**
1. Find all pending invitations for this email: `getPrismaClient().invitation.findMany({ where: { email, status: 'pending' } })`
2. For each pending invitation:
   - Check if UserTenantRole already exists for this userId+tenantId
   - If not, create UserTenantRole with the invitation's role
   - Update invitation status to 'accepted'
3. This is called from the session callback in auth-options.ts

**1. Create `web-ui/app/api/invitations/route.ts`** — POST (create invitation) + GET (list invitations):

```typescript
import { NextRequest, NextResponse } from "next/server";
import { authorize } from "@/lib/rbac/authorize";
import { getSessionTenantId, getAuthSession } from "@/lib/auth-session";
import { InvitationService } from "@/lib/invitation-service";

export async function POST(request: NextRequest) {
    console.log("API - POST /api/invitations - Creating invitation");
    // Per INVT-01: admin can invite users — authorize 'create' on 'User' (maps to Settings module)
    const authError = await authorize("create", "User");
    if (authError) return authError;

    try {
        const tenantId = await getSessionTenantId();
        const session = await getAuthSession();
        const invitedBy = session!.user.id;
        const { email, role } = await request.json();

        if (!email || !role) {
            return NextResponse.json({ success: false, error: "Email and role are required" }, { status: 400 });
        }

        const result = await InvitationService.createInvitation(tenantId, email, role, invitedBy);
        return NextResponse.json({ success: true, data: result }, { status: 201 });
    } catch (error) {
        console.error("API - Error creating invitation:", error);
        return NextResponse.json({
            success: false,
            error: error instanceof Error ? error.message : "Failed to create invitation"
        }, { status: error instanceof Error && error.message.includes("already") ? 409 : 500 });
    }
}

export async function GET() {
    console.log("API - GET /api/invitations - Listing invitations");
    const authError = await authorize("read", "User");
    if (authError) return authError;

    try {
        const tenantId = await getSessionTenantId();
        const invitations = await InvitationService.listInvitations(tenantId);
        return NextResponse.json({ success: true, data: invitations });
    } catch (error) {
        console.error("API - Error listing invitations:", error);
        return NextResponse.json({
            success: false,
            error: error instanceof Error ? error.message : "Failed to list invitations"
        }, { status: 500 });
    }
}
```
